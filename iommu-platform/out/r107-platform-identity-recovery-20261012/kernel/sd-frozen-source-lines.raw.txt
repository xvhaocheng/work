   143	{
   144	    volatile uint8_t *p = dst;
   145	    long i = size;
   146	    int rc = 0;

   154	    crc = crc7(crc, (src_lba >> 24) & 0xff);
   155	    crc = crc7(crc, (src_lba >> 16) & 0xff);
   156	    crc = crc7(crc, (src_lba >> 8) & 0xff);
   157	    crc = crc7(crc, src_lba & 0xff);
   158	    crc = (crc << 1) | 1;
   159	    if (sd_cmd(18, src_lba, crc) != 0x00)

    56	    for (int i = 0; i < 100; i++)
    57	    {
    58	        __asm__ volatile(

    66	        ;
    67

    81	    write_reg(SPI_CONTROL_REG, 0x06);

    86	int spi_write_bytes(uint8_t *bytes, uint32_t len, uint8_t *ret)
    87	{
    88	    uint32_t status;
    89	
    90	    if (len > 256) // FIFO maxdepth 256
    91	        return -1;
    92	
    93	    // enable slave select
    94	    write_reg(SPI_SLAVE_SELECT_REG, 0xfffffffe);
    95	
    96	    for (int i = 0; i < len; i++)
    97	    {
    98	        write_reg(SPI_TRANSMIT_REG, bytes[i] & 0xff);
    99	    }
   100	
   101	    for (int i = 0; i < 50; i++)
   102	    {
   103	        __asm__ volatile(
   104	            "nop");
   105	    }
   106	
   107	    // enable spi control master flag
   108	    write_reg(SPI_CONTROL_REG, 0x106);
   109	
   110	    do
   111	    {
   112	        status = read_reg(SPI_STATUS_REG);
   113	    } while ((status & 0x1) == 0x1);
   114	
   115	    for (int i = 0; i < len;)
   116	    {
   117	        status = read_reg(SPI_STATUS_REG);
   118	        if ((status & 0x1) != 0x1) // recieve fifo not empty
   119	        {
   120	            ret[i++] = read_reg(SPI_RECEIVE_REG);
   121	        }
   122	    }
   123	
   124	    // disable slave select
   125	    write_reg(SPI_SLAVE_SELECT_REG, 0xffffffff);
   126	
   127	    // disable spi control master flag
   128	    write_reg(SPI_CONTROL_REG, 0x06);
   129	
   130	    return 0;
