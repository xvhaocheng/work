    76						unsigned shift, bool root,
    77						size_t pgsize,
    78						unsigned long (*pd_alloc)(gfp_t),
    79						gfp_t gfp)
    80	{
    81		pmd_t *pte;
    82		unsigned long pfn;
    83	
    84		if (root)
    85			pte = ptp + ((iova >> shift) & (PTRS_PER_PMD - 1));
    86		else
    87			pte = (pmd_t *) pfn_to_virt(__page_val_to_pfn(pmd_val(*ptp))) +
    88			    ((iova >> shift) & (PTRS_PER_PMD - 1));
    89	
    90		if ((1ULL << shift) <= pgsize) {
    91			if (pmd_present(*pte) && !pmd_leaf(*pte))
    92				riscv_iommu_pt_walk_free(pte, shift - 9, false);
    93			return (pte_t *) pte;
    94		}
    95	
    96		if (pmd_none(*pte)) {
    97			pfn = pd_alloc ? virt_to_pfn(pd_alloc(gfp)) : 0;
    98			if (!pfn)
    99				return NULL;
   100			set_pmd(pte, __pmd((pfn << _PAGE_PFN_SHIFT) | _PAGE_TABLE));
   101		}
   102	
   103		return riscv_iommu_pt_walk_alloc(pte, iova, shift - 9, false,
   104						 pgsize, pd_alloc, gfp);
   105	}

   107	static pte_t *riscv_iommu_pt_walk_fetch(pmd_t * ptp,
   108						unsigned long iova, unsigned shift,
   109						bool root, unsigned int *leaf_shift)
   110	{
   111		pmd_t *pte;
   112	
   113		if (root)
   114			pte = ptp + ((iova >> shift) & (PTRS_PER_PMD - 1));
   115		else
   116			pte = (pmd_t *) pfn_to_virt(__page_val_to_pfn(pmd_val(*ptp))) +
   117			    ((iova >> shift) & (PTRS_PER_PMD - 1));
   118	
   119		if (pmd_leaf(*pte)) {
   120			if (leaf_shift)
   121				*leaf_shift = shift;
   122			return (pte_t *) pte;
   123		} else if (pmd_none(*pte))
   124			return NULL;
   125		else if (shift == PAGE_SHIFT)
   126			return NULL;
   127	
   128		return riscv_iommu_pt_walk_fetch(pte, iova, shift - 9, false, leaf_shift);
   129	}

   209	static phys_addr_t riscv_iommu_iova_to_phys(struct io_pgtable_ops *ops,
   210						    unsigned long iova)
   211	{
   212		struct riscv_iommu_domain *domain = io_pgtable_ops_to_domain(ops);
   213		unsigned int leaf_shift;
   214		pte_t *pte;
   215	
   216		if (domain->domain.type == IOMMU_DOMAIN_IDENTITY)
   217			return (phys_addr_t) iova;
   218	
   219		pte = riscv_iommu_pt_walk_fetch((pmd_t *) domain->pgd_root,
   220						iova, PGDIR_SHIFT, true, &leaf_shift);
   221		if (!pte || !pte_present(*pte))
   222			return 0;
   223	
   224		/* Add the offset within the actual leaf (4 KiB, 2 MiB or 1 GiB). */
   225		return pfn_to_phys(pte_pfn(*pte)) | (iova & ((1UL << leaf_shift) - 1));
   226	}
