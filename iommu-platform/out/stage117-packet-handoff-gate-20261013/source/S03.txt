173: vhost_async_dma_check_completed(struct virtio_net *dev, int16_t dma_id, uint16_t vchan_id,
174: 		uint16_t max_pkts)
175: {
176: 	struct async_dma_vchan_info *dma_info = &dma_copy_track[dma_id].vchans[vchan_id];
177: 	uint16_t ring_mask = dma_info->ring_mask;
178: 	uint16_t last_idx = 0;
179: 	uint16_t nr_copies;
180: 	uint16_t copy_idx;
181: 	uint16_t i;
182: 	bool has_error = false;
183: 	static bool vhost_async_dma_complete_log;
184: 
185: 	rte_spinlock_lock(&dma_info->dma_lock);
186: 
187: 	/**
188: 	 * Print error log for debugging, if DMA reports error during
189: 	 * DMA transfer. We do not handle error in vhost level.
190: 	 */
191: 	nr_copies = rte_dma_completed(dma_id, vchan_id, max_pkts, &last_idx, &has_error);
192: 	if (unlikely(!vhost_async_dma_complete_log && has_error)) {
193: 		VHOST_DATA_LOG(dev->ifname, ERR,
194: 			"DMA completion failure on channel %d:%u",
195: 			dma_id, vchan_id);
196: 		vhost_async_dma_complete_log = true;
197: 	} else if (nr_copies == 0) {
198: 		goto out;
199: 	}
200: 
201: 	copy_idx = last_idx - nr_copies + 1;
202: 	for (i = 0; i < nr_copies; i++) {
203: 		bool *flag;
204: 
205: 		flag = dma_info->pkts_cmpl_flag_addr[copy_idx & ring_mask];
206: 		if (flag) {
207: 			/**
208: 			 * Mark the packet flag as received. The flag
209: 			 * could belong to another virtqueue but write
210: 			 * is atomic.
211: 			 */
212: 			*flag = true;
213: 			dma_info->pkts_cmpl_flag_addr[copy_idx & ring_mask] = NULL;
214: 		}
215: 		copy_idx++;
216: 	}
217: 
218: out:
219: 	rte_spinlock_unlock(&dma_info->dma_lock);
220: 	return nr_copies;
