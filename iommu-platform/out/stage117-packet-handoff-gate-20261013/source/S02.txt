91: vhost_async_dma_transfer_one(struct virtio_net *dev, struct vhost_virtqueue *vq,
92: 		int16_t dma_id, uint16_t vchan_id, uint16_t flag_idx,
93: 		struct vhost_iov_iter *pkt)
94: 	__rte_requires_shared_capability(&vq->access_lock)
95: {
96: 	struct async_dma_vchan_info *dma_info = &dma_copy_track[dma_id].vchans[vchan_id];
97: 	uint16_t ring_mask = dma_info->ring_mask;
98: 	static bool vhost_async_dma_copy_log;
99: 
100: 
101: 	struct vhost_iovec *iov = pkt->iov;
102: 	int copy_idx = 0;
103: 	uint32_t nr_segs = pkt->nr_segs;
104: 	uint16_t i;
105: 
106: 	if (rte_dma_burst_capacity(dma_id, vchan_id) < nr_segs)
107: 		return -1;
108: 
109: 	for (i = 0; i < nr_segs; i++) {
110: 		copy_idx = rte_dma_copy(dma_id, vchan_id, (rte_iova_t)iov[i].src_addr,
111: 				(rte_iova_t)iov[i].dst_addr, iov[i].len, RTE_DMA_OP_FLAG_LLC);
112: 		/**
113: 		 * Since all memory is pinned and DMA vChannel
114: 		 * ring has enough space, failure should be a
115: 		 * rare case. If failure happens, it means DMA
116: 		 * device encounters serious errors; in this
117: 		 * case, please stop async data-path and check
118: 		 * what has happened to DMA device.
119: 		 */
120: 		if (unlikely(copy_idx < 0)) {
121: 			if (!vhost_async_dma_copy_log) {
122: 				VHOST_DATA_LOG(dev->ifname, ERR,
123: 					"DMA copy failed for channel %d:%u",
124: 					dma_id, vchan_id);
125: 				vhost_async_dma_copy_log = true;
126: 			}
127: 			return -1;
128: 		}
129: 	}
130: 
131: 	/**
132: 	 * Only store packet completion flag address in the last copy's
133: 	 * slot, and other slots are set to NULL.
134: 	 */
135: 	dma_info->pkts_cmpl_flag_addr[copy_idx & ring_mask] = &vq->async->pkts_cmpl_flag[flag_idx];
136: 
137: 	return nr_segs;
138: }
