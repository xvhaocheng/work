`default_nettype none
//=============================================================================
// dma_research_policy.sv
//
// Standalone research "Method control" policy block (bank E00..E5F).
// NEW FILE - no existing RTL is modified.
//
// Implements docs/research-two-bit-v2.md "方法控制（独立模块，bank E00起）".
//
// Register map (index_i[7:0]; index_i[12:8] must be 0)
// ---------------------------------------------------------------------------
//  E00 r  magic     0x49444d414d435431  ("IDMAMCT1")
//  E01 r  version   1
//  E02 r  error     sticky, see ERR_* bits
//  E03 r  status    bit0 armed / bit1 frozen / bit2 abort
//  E04 w  command   1 ARM / 2 ABORT / 3 FREEZE / 4 CLEAR   (reads 0)
//  E05 rw mode                    default 0
//  E06 rw window_cycles           default 16384
//  E07 rw target_bytes            default 51540
//  E08 rw deadband_bytes          default 4096
//  E09 rw initial_budget          default 8000
//  E0A rw min_budget              default 1
//  E0B rw max_budget              default 32768
//  E0C rw budget_step             default 512
//  E0D rw credit_cap              default 65536
//  E0E rw hit_weight              default 1
//  E0F rw miss_weight             default 100
//  E10 rw ptw_weight              default 100
//  --- statistics (read only, CounterWidth wide) -----------------------------
//  E20 complete (fully V-active) billing windows
//  E21 V done events
//  E22 V done bytes (4096 per done)
//  E23 windows whose observed bytes < target
//  E24 total shortfall bytes (target - observed)
//  E25 cycles with policy_enable_o=1 and policy_allow_o=0
//  E26 total charge
//  E27 total replenish
//  E28 A issue events
//  E29 A done events
//  E2A cycles with non-zero charge
//  E2B A-owned lookup hit events
//  E2C A-owned lookup miss events
//  E2D A-owned PTW return events (ctl_did==DID_A && ptw_r_fire)
//  E2E cycles with ctl_did==DID_A (A-owner cycles)
//  E2F A select fires
//  E30 A-owned lookup events (hit+miss)
//  E31 A-owned lookup miss events
//  E32 A REQ_TRANS/AWAIT_RESP cycles
//  E33 window boundaries closed
//  E34 window boundaries that were NOT fully V-active
//  E35 feedback steps that increased b
//  E36 feedback steps that decreased b
//  E37 mode6 gap loads (A done after the first copy)
//  E38 mode7 stale charges applied
//  E39 cycles with armed=1 and frozen=0
//  E3A final signed balance C (two's complement, 64 bit)
//  E3B minimum b seen
//  E3C maximum b seen
//  E3D final b
//  E3E longest V-active run with no V done (cycles, 64 bit)
//  E3F reserved, reads 0
//  E40..E5F V-done-per-complete-active-window histogram, 32 buckets:
//          bucket k (k=0..30) = windows with k V done, bucket 31 = ">=31"
// ---------------------------------------------------------------------------
//  Cost per mode:
//    0 unlimited/no charge | 1 A issue = 4096 B | 2 A select = 1
//    3 A lookup miss = 1   | 4 weighted (hit/miss/ptw) | 5 A REQ_TRANS/
//    AWAIT_RESP cycle = 1  | 6 V-only feedback gap (no per-cycle charge)
//    7 stale actual (previous complete window A-owner cycles at window end)
//=============================================================================

module dma_research_policy #(
    parameter bit          Enabled      = 1'b1,
    parameter int unsigned CounterWidth = 64,
    parameter logic [23:0] DID_A        = 24'h000010,
    parameter logic [2:0]  ST_REQ_TRANS = 3'd1,
    parameter logic [2:0]  ST_AWAIT_RESP= 3'd2
)(
    input  logic        clk_i,
    input  logic        rst_ni,

    input  logic        cfg_valid_i,
    input  logic        cfg_write_i,
    input  logic [12:0] index_i,
    input  logic [63:0] wdata_i,
    output logic [63:0] rdata_o,

    input  logic        quiet_i,
    input  logic        supported_i,

    input  logic        v_active_i,
    input  logic        a_issue_i,
    input  logic        a_done_i,
    input  logic        v_done_i,
    input  logic [2:0]  ctl_state_i,
    input  logic [23:0] ctl_did_i,
    input  logic        select_fire_i,
    input  logic [23:0] select_did_i,
    input  logic        lookup_i,
    input  logic        lookup_hit_i,
    input  logic [23:0] lookup_did_i,
    input  logic        ptw_r_fire_i,
    input  logic        response_error_i,

    output logic        policy_enable_o,
    output logic        policy_allow_o,
    output logic        policy_abort_o
);

    //=========================================================================
    // 1. Constants, register offsets, error bits, stat indices
    //=========================================================================
    localparam logic [63:0] MAGIC      = 64'h49444d414d435431; // "IDMAMCT1"
    localparam logic [63:0] VERSION    = 64'd1;
    localparam logic [63:0] BYTES_DONE = 64'd4096;

    localparam logic [7:0] R_MAGIC    = 8'h00;
    localparam logic [7:0] R_VERSION  = 8'h01;
    localparam logic [7:0] R_ERROR    = 8'h02;
    localparam logic [7:0] R_STATUS   = 8'h03;
    localparam logic [7:0] R_COMMAND  = 8'h04;
    localparam logic [7:0] R_MODE     = 8'h05;
    localparam logic [7:0] R_WINDOW   = 8'h06;
    localparam logic [7:0] R_TARGET   = 8'h07;
    localparam logic [7:0] R_DEADBAND = 8'h08;
    localparam logic [7:0] R_INITBUD  = 8'h09;
    localparam logic [7:0] R_MINBUD   = 8'h0A;
    localparam logic [7:0] R_MAXBUD   = 8'h0B;
    localparam logic [7:0] R_STEP     = 8'h0C;
    localparam logic [7:0] R_CAP      = 8'h0D;
    localparam logic [7:0] R_WHIT     = 8'h0E;
    localparam logic [7:0] R_WMISS    = 8'h0F;
    localparam logic [7:0] R_WPTW     = 8'h10;
    localparam logic [7:0] R_LASTWR   = 8'h10;

    localparam logic [7:0] S_CNT_BASE  = 8'h20; // E20..E39
    localparam logic [7:0] S_BALANCE   = 8'h3A;
    localparam logic [7:0] S_BMIN      = 8'h3B;
    localparam logic [7:0] S_BMAX      = 8'h3C;
    localparam logic [7:0] S_BFINAL    = 8'h3D;
    localparam logic [7:0] S_NODONE    = 8'h3E;
    localparam logic [7:0] S_HIST_BASE = 8'h40;
    localparam logic [7:0] S_HIST_LAST = 8'h5F;

    // sticky error bits (E02)
    localparam int E_BADADDR  = 0;  // bad address / high bits nonzero on write
    localparam int E_BADCMD   = 1;  // unknown command code
    localparam int E_UNSUP    = 2;  // ARM while !supported_i or Enabled=0
    localparam int E_CFGLOCK  = 3;  // config write while armed
    localparam int E_CNTOVF   = 4;  // statistics counter overflow (fatal)
    localparam int E_BALOVF   = 5;  // signed 64 balance overflow (fatal)
    localparam int E_CLEARILL = 6;  // illegal CLEAR (armed or not quiet)
    localparam int E_FRZILL   = 7;  // FREEZE while not quiet
    localparam int E_ARMSTATE = 8;  // ARM/ABORT/FREEZE in a wrong state
    localparam int E_OBS      = 9;  // observation error while armed (fatal)
    localparam int E_BADCFG   = 10; // config does not satisfy ARM predicates

    localparam logic [63:0] FATAL_MASK =
        (64'd1 << E_CNTOVF) | (64'd1 << E_BALOVF) | (64'd1 << E_OBS);

    // statistics counter indices -> E20..E39
    localparam int ST_WIN       = 0;  // E20
    localparam int ST_VDONE     = 1;  // E21
    localparam int ST_VBYTES    = 2;  // E22
    localparam int ST_SHORTW    = 3;  // E23
    localparam int ST_SHORTB    = 4;  // E24
    localparam int ST_BLOCKED   = 5;  // E25
    localparam int ST_CHARGE    = 6;  // E26
    localparam int ST_REPLEN    = 7;  // E27
    localparam int ST_AISSUE    = 8;  // E28
    localparam int ST_ADONE     = 9;  // E29
    localparam int ST_CHARGECYC = 10; // E2A
    localparam int ST_HIT       = 11; // E2B
    localparam int ST_MISS      = 12; // E2C
    localparam int ST_PTW       = 13; // E2D
    localparam int ST_AOWN      = 14; // E2E
    localparam int ST_SEL       = 15; // E2F
    localparam int ST_LUTA      = 16; // E30
    localparam int ST_LUTAMISS  = 17; // E31
    localparam int ST_ACTUALCYC = 18; // E32
    localparam int ST_WINALL    = 19; // E33
    localparam int ST_WINPART   = 20; // E34
    localparam int ST_FBUP      = 21; // E35
    localparam int ST_FBDN      = 22; // E36
    localparam int ST_GAPL      = 23; // E37
    localparam int ST_STALE     = 24; // E38
    localparam int ST_ARMEDCYC  = 25; // E39
    localparam int NSTAT        = 26;

    localparam int NHIST = 32;

    // reset-time defaults (contract "初次defaultconfig")
    localparam logic [31:0] DEF_MODE     = 32'd0;
    localparam logic [31:0] DEF_WINDOW   = 32'd16384;
    localparam logic [31:0] DEF_TARGET   = 32'd51540;
    localparam logic [31:0] DEF_DEADBAND = 32'd4096;
    localparam logic [31:0] DEF_INITBUD  = 32'd8000;
    localparam logic [31:0] DEF_MINBUD   = 32'd1;
    localparam logic [31:0] DEF_MAXBUD   = 32'd32768;
    localparam logic [31:0] DEF_STEP     = 32'd512;
    localparam logic [31:0] DEF_CAP      = 32'd65536;
    localparam logic [31:0] DEF_WHIT     = 32'd1;
    localparam logic [31:0] DEF_WMISS    = 32'd100;
    localparam logic [31:0] DEF_WPTW     = 32'd100;

    //=========================================================================
    // 2. State
    //=========================================================================
    logic        armed_q, frozen_q, aborted_q;
    logic [63:0] err_q;

    logic [31:0] mode_q, window_q, target_q, deadband_q, initbud_q,
                 minbud_q, maxbud_q, step_q, cap_q,
                 whit_q, wmiss_q, wptw_q;

    logic signed [63:0] balance_q;
    logic [63:0]        b_q, bmin_q, bmax_q;
    logic [63:0]        prev_main_q, win_main_q;
    logic [63:0]        win_cnt_q, win_vdone_q;
    logic               win_allact_q;
    logic [63:0]        nodone_run_q, nodone_max_q;
    logic               m6_started_q;
    logic [63:0]        m6_cd_q;

    logic [CounterWidth-1:0] stat_q [0:NSTAT-1];
    logic [CounterWidth-1:0] hist_q [0:NHIST-1];

    logic [63:0] stat_inc [0:NSTAT-1];
    logic [64:0] stat_sum [0:NSTAT-1];
    logic [63:0] hist_inc [0:NHIST-1];
    logic [64:0] hist_sum [0:NHIST-1];

    //=========================================================================
    // 3. Helper functions
    //=========================================================================
    // (CounterWidth + 1)-bit sum plus carry-out at bit CounterWidth.
    function automatic logic [64:0] cadd(input logic [CounterWidth-1:0] c,
                                         input logic [63:0]          inc);
        logic [64:0] w;
        begin
            w    = {1'b0, c} + {1'b0, inc};
            cadd = w;
        end
    endfunction

    // zero-extend a CounterWidth value to 64 bit (portable, no zero replication)
    function automatic logic [63:0] zext64(input logic [CounterWidth-1:0] v);
        logic [63:0] r;
        begin
            r = 64'd0;
            for (int i = 0; i < CounterWidth; i++) r[i] = v[i];
            zext64 = r;
        end
    endfunction

    //=========================================================================
    // 4. Event qualifiers (all in the peripheral clk domain)
    //=========================================================================
    logic ev_sel_a, ev_lut_a, ev_lut_hit_a, ev_lut_miss_a, ev_ptw_a,
          ev_actual_a, ev_aown;

    assign ev_sel_a      = select_fire_i && (select_did_i == DID_A);
    assign ev_lut_a      = lookup_i      && (lookup_did_i == DID_A);
    assign ev_lut_hit_a  = ev_lut_a &&  lookup_hit_i;
    assign ev_lut_miss_a = ev_lut_a && !lookup_hit_i;
    assign ev_ptw_a      = ptw_r_fire_i  && (ctl_did_i == DID_A);
    assign ev_actual_a   = (ctl_did_i == DID_A) &&
                           ((ctl_state_i == ST_REQ_TRANS) || (ctl_state_i == ST_AWAIT_RESP));
    assign ev_aown       = (ctl_did_i == DID_A);

    //=========================================================================
    // 5. Window control
    //=========================================================================
    logic        win_run, win_last, act_all_now, win_complete;
    logic [63:0] vdone_now, vdone_bytes_now, wnd_m1;

    assign wnd_m1       = {32'd0, window_q} - 64'd1;
    assign win_run      = armed_q & ~frozen_q;
    assign win_last     = win_run & (window_q != 32'd0) & (win_cnt_q == wnd_m1);
    assign act_all_now  = win_allact_q & v_active_i;   // includes first and last cycle
    assign vdone_now    = win_vdone_q + (v_done_i ? 64'd1 : 64'd0);
    assign win_complete = win_last & act_all_now;

    assign vdone_bytes_now = vdone_now * BYTES_DONE;

    //=========================================================================
    // 6. Cost / charge / balance
    //=========================================================================
    logic [63:0] cost_ev, charge_now, refill_now, main_inc, shortfall_bytes;
    logic        stale_cyc, refill_en;

    always_comb begin
        cost_ev = 64'd0;
        case (mode_q)
            32'd0: cost_ev = 64'd0;
            32'd1: cost_ev = a_issue_i ? BYTES_DONE : 64'd0;
            32'd2: cost_ev = ev_sel_a  ? 64'd1 : 64'd0;
            32'd3: cost_ev = ev_lut_miss_a ? 64'd1 : 64'd0;
            32'd4: cost_ev = (ev_lut_hit_a  ? {32'd0, whit_q}  : 64'd0)
                           + (ev_lut_miss_a ? {32'd0, wmiss_q} : 64'd0)
                           + (ev_ptw_a      ? {32'd0, wptw_q}  : 64'd0);
            32'd5: cost_ev = ev_actual_a ? 64'd1 : 64'd0;
            default: cost_ev = 64'd0;   // 6 and 7 carry no per-cycle cost
        endcase
    end

    assign stale_cyc = win_last & (mode_q == 32'd7);
    assign charge_now = (win_run ? cost_ev : 64'd0) + (stale_cyc ? prev_main_q : 64'd0);

    // mode7 "main" = A-owner cycles of the open window
    assign main_inc = (win_run & (mode_q == 32'd7) & ev_aown) ? 64'd1 : 64'd0;

    assign refill_en  = win_complete & (mode_q != 32'd0) & (mode_q != 32'd6);
    assign refill_now = refill_en ? b_q : 64'd0;

    logic signed [65:0] bal_s, bal_c, bal_r, cap_s;
    logic               bal_ovf;
    logic signed [63:0] bal_next;

    assign bal_s = {{2{balance_q[63]}}, balance_q};
    assign bal_c = bal_s - $signed({2'b00, charge_now});
    assign bal_r = bal_c + $signed({2'b00, refill_now});
    assign cap_s = $signed({2'b00, cap_q});

    always_comb begin
        // positive side only is capped by credit_cap; debt is never forgiven
        if (refill_en && (bal_r > cap_s)) bal_r_clamped = cap_s;
        else                              bal_r_clamped = bal_r;
    end
    logic signed [65:0] bal_r_clamped;

    assign bal_ovf  = (bal_r_clamped[65:63] != 3'b000) &&
                      (bal_r_clamped[65:63] != 3'b111);
    assign bal_next = bal_ovf
                    ? (bal_r_clamped[65] ? {1'b1, {63{1'b0}}} : {1'b0, {63{1'b1}}})
                    : bal_r_clamped[63:0];

    //=========================================================================
    // 7. Adaptive feedback (only on complete, fully V-active windows)
    //=========================================================================
    logic signed [65:0] obs_s, tgt_s, db_s;
    logic               fb_inc, fb_dec;
    logic [63:0]        b_next, b_inc_val, b_dec_val;

    assign obs_s = $signed({2'b00, vdone_bytes_now});
    assign tgt_s = $signed({34'd0, target_q});
    assign db_s  = $signed({34'd0, deadband_q});

    always_comb begin
        // billing modes: observed above target -> b up; below -> b down
        // mode 6 is reversed: V slow -> gap up
        if (mode_q == 32'd6) begin
            fb_inc = (obs_s + db_s < tgt_s);
            fb_dec = (obs_s > tgt_s + db_s);
        end else begin
            fb_inc = (obs_s > tgt_s + db_s);
            fb_dec = (obs_s + db_s < tgt_s);
        end
    end

    logic signed [64:0] b_up_s, b_dn_s;
    assign b_up_s = $signed({1'b0, b_q}) + $signed({33'd0, step_q});
    assign b_dn_s = $signed({1'b0, b_q}) - $signed({33'd0, step_q});
    assign b_inc_val = (b_up_s > $signed({33'd0, maxbud_q})) ? {32'd0, maxbud_q}
                                                             : b_up_s[63:0];
    assign b_dec_val = (b_dn_s < $signed({33'd0, minbud_q})) ? {32'd0, minbud_q}
                                                             : b_dn_s[63:0];
    assign b_next    = fb_inc ? b_inc_val : (fb_dec ? b_dec_val : b_q);

    assign shortfall_bytes = {32'd0, target_q} - vdone_bytes_now;

    //=========================================================================
    // 8. mode6 gap countdown (decision uses the updated countdown)
    //=========================================================================
    logic [63:0] m6_cd_dec, m6_cd_next;
    logic        gap_load;

    assign m6_cd_dec = (win_run && (m6_cd_q != 64'd0)) ? (m6_cd_q - 64'd1) : m6_cd_q;

    always_comb begin
        m6_cd_next = m6_cd_dec;
        gap_load   = 1'b0;
        if (win_run && (mode_q == 32'd6) && a_done_i && m6_started_q) begin
            m6_cd_next = b_q;      // load current gap
            gap_load   = 1'b1;
        end
    end
    // first A done never inserts a gap (m6_started_q is still 0 that cycle)

    //=========================================================================
    // 9. Statistics increments and overflow
    //=========================================================================
    logic [4:0] hist_idx;
    assign hist_idx = (vdone_now >= 64'd31) ? 5'd31 : vdone_now[4:0];

    always_comb begin
        for (int i = 0; i < NSTAT; i++) stat_inc[i] = 64'd0;
        for (int i = 0; i < NHIST; i++) hist_inc[i] = 64'd0;

        stat_inc[ST_ARMEDCYC]  = win_run ? 64'd1 : 64'd0;
        stat_inc[ST_BLOCKED]   = admit_blocked ? 64'd1 : 64'd0;
        stat_inc[ST_CHARGE]    = charge_now;
        stat_inc[ST_CHARGECYC] = (charge_now != 64'd0) ? 64'd1 : 64'd0;
        stat_inc[ST_REPLEN]    = refill_now;

        stat_inc[ST_VDONE]     = (win_run && v_done_i)  ? 64'd1 : 64'd0;
        stat_inc[ST_VBYTES]    = (win_run && v_done_i)  ? BYTES_DONE : 64'd0;
        stat_inc[ST_AISSUE]    = (win_run && a_issue_i) ? 64'd1 : 64'd0;
        stat_inc[ST_ADONE]     = (win_run && a_done_i)  ? 64'd1 : 64'd0;
        stat_inc[ST_HIT]       = (win_run && ev_lut_hit_a)  ? 64'd1 : 64'd0;
        stat_inc[ST_MISS]      = (win_run && ev_lut_miss_a) ? 64'd1 : 64'd0;
        stat_inc[ST_PTW]       = (win_run && ev_ptw_a)      ? 64'd1 : 64'd0;
        stat_inc[ST_AOWN]      = (win_run && ev_aown)       ? 64'd1 : 64'd0;
        stat_inc[ST_SEL]       = (win_run && ev_sel_a)      ? 64'd1 : 64'd0;
        stat_inc[ST_LUTA]      = (win_run && ev_lut_a)      ? 64'd1 : 64'd0;
        stat_inc[ST_LUTAMISS]  = (win_run && ev_lut_miss_a) ? 64'd1 : 64'd0;
        stat_inc[ST_ACTUALCYC] = (win_run && ev_actual_a)   ? 64'd1 : 64'd0;

        stat_inc[ST_WIN]     = win_complete ? 64'd1 : 64'd0;
        stat_inc[ST_WINALL]  = win_last ? 64'd1 : 64'd0;
        stat_inc[ST_WINPART] = (win_last && !act_all_now) ? 64'd1 : 64'd0;
        stat_inc[ST_GAPL]    = gap_load ? 64'd1 : 64'd0;
        stat_inc[ST_STALE]   = stale_cyc ? 64'd1 : 64'd0;

        stat_inc[ST_SHORTW]  = (win_complete && (obs_s < tgt_s)) ? 64'd1 : 64'd0;
        stat_inc[ST_SHORTB]  = (win_complete && (obs_s < tgt_s)) ? shortfall_bytes
                                                                 : 64'd0;
        stat_inc[ST_FBUP]    = (win_complete && (mode_q != 32'd0) && fb_inc) ? 64'd1 : 64'd0;
        stat_inc[ST_FBDN]    = (win_complete && (mode_q != 32'd0) && fb_dec) ? 64'd1 : 64'd0;

        if (win_complete) hist_inc[hist_idx] = 64'd1;
    end

    logic cnt_ovf;

    always_comb begin
        cnt_ovf = 1'b0;
        for (int i = 0; i < NSTAT; i++) stat_sum[i] = cadd(stat_q[i], stat_inc[i]);
        for (int i = 0; i < NHIST; i++) hist_sum[i] = cadd(hist_q[i], hist_inc[i]);
        for (int i = 0; i < NSTAT; i++) if (stat_sum[i][CounterWidth]) cnt_ovf = 1'b1;
        for (int i = 0; i < NHIST; i++) if (hist_sum[i][CounterWidth]) cnt_ovf = 1'b1;
    end

    //=========================================================================

        prev_main = win_main = 0;
        win_cnt = win_vdone = 0;
        win_allact = true;
        m6_started = false; m6_cd = 0;
        nodone_run = 0;
    }

    void set_cfg(uint32_t idx, uint32_t v) {
        switch (idx) {
            case R_MODE:     mode = v; break;
            case R_WINDOW:   window = v; break;
            case R_TARGET:   target = v; break;
            case R_DEADBAND: deadband = v; break;
            case R_INITBUD:  initbud = v; break;
            case R_MINBUD:   minbud = v; break;
            case R_MAXBUD:   maxbud = v; break;
            case R_STEP:     step = v; break;
            case R_CAP:      cap = v; break;
            case R_WHIT:     whit = v; break;
            case R_WMISS:    wmiss = v; break;
            case R_WPTW:     wptw = v; break;
            default: break;
        }
    }

    // Fills the write side-effect flags; returns the error bits raised by the write.
    uint64_t decode_write(uint32_t idx, uint64_t data, bool quiet,
                          bool &f_arm, bool &f_abort, bool &f_freeze,
                          bool &f_clear, bool &f_cfg) {
        uint64_t ne = 0;
        f_arm = f_abort = f_freeze = f_clear = f_cfg = false;

        bool ok = (idx >= R_COMMAND) && (idx <= R_WPTW) && ((data >> 32) == 0);
        if (!ok) { ne |= (1ull << E_BADADDR); return ne; }

        if (idx == R_COMMAND) {
            uint32_t cmd = (uint32_t)data;
            if (cmd == 1) {                                   // ARM
                if (!enabled)                       ne |= (1ull << E_UNSUP);
                else if (armed || frozen || !quiet) ne |= (1ull << E_ARMSTATE);
                else if (config_bad())              ne |= (1ull << E_BADCFG);
                else                                f_arm = true;
            } else if (cmd == 2) {                            // ABORT
                if (!armed || frozen)               ne |= (1ull << E_ARMSTATE);
                else if (!enabled)                  ne |= (1ull << E_UNSUP);
                else                                f_abort = true;
            } else if (cmd == 3) {                            // FREEZE
                if (!armed || frozen)               ne |= (1ull << E_ARMSTATE);
                else if (!quiet)                    ne |= (1ull << E_FRZILL);
                else                                f_freeze = true;
            } else if (cmd == 4) {                            // CLEAR
                if (!quiet || armed)                ne |= (1ull << E_CLEARILL);
                else                                f_clear = true;
            } else {
                ne |= (1ull << E_BADCMD);
            }
        } else {
            f_cfg = true;
            if (armed) ne |= (1ull << E_CFGLOCK);
        }
        return ne;
    }

    RefObs cycle(const In &in, bool do_wr, uint32_t widx, uint64_t wdata) {
        // ---- events / cost -------------------------------------------------
        bool ev_sel_a  = in.select_fire && in.select_did == did_a;
        bool ev_lut_a  = in.lookup && in.lookup_did == did_a;
        bool ev_hit_a  = ev_lut_a && in.lookup_hit;
        bool ev_miss_a = ev_lut_a && !in.lookup_hit;
        bool ev_ptw_a  = in.ptw_r_fire && in.ctl_did == did_a;
        bool ev_act_a  = (in.ctl_did == did_a) && (in.ctl_state == 1 || in.ctl_state == 2);
        bool ev_aown   = (in.ctl_did == did_a);

        uint64_t cost = 0;
        switch (mode) {
            case 0: cost = 0; break;
            case 1: cost = in.a_issue ? 4096 : 0; break;
            case 2: cost = ev_sel_a ? 1 : 0; break;
            case 3: cost = ev_miss_a ? 1 : 0; break;
            case 4: cost = (ev_hit_a ? whit : 0) + (ev_miss_a ? wmiss : 0)
                         + (ev_ptw_a ? wptw : 0); break;
            case 5: cost = ev_act_a ? 1 : 0; break;
            default: cost = 0; break;
        }

        // ---- window --------------------------------------------------------
        bool win_run  = armed && !frozen;
        bool win_last = win_run && window != 0 && win_cnt == (uint64_t)window - 1;
        bool act_all  = win_allact && in.v_active;
        uint64_t vdone_now = win_vdone + (in.v_done ? 1 : 0);
        bool win_complete = win_last && act_all;
        bool stale = win_last && mode == 7;
        uint64_t charge_now = (win_run ? cost : 0) + (stale ? prev_main : 0);
        bool refill_en = win_complete && mode != 0 && mode != 6;
        uint64_t refill_now = refill_en ? b : 0;

        // ---- balance (66-bit intermediate, debt preserved) -----------------
        __int128 balr = (__int128)balance - (__int128)charge_now + (__int128)refill_now;
        if (refill_en && balr > (__int128)cap) balr = (__int128)cap;
        bool bal_ovf = (balr > (__int128)INT64_MAX) || (balr < (__int128)INT64_MIN);
        int64_t bal_next = bal_ovf ? (balr > 0 ? INT64_MAX : INT64_MIN) : (int64_t)balr;

        // ---- feedback ------------------------------------------------------
        __int128 obs = (__int128)(vdone_now * 4096ull);
        __int128 tgt = (__int128)target;
        __int128 db  = (__int128)deadband;
        bool up_raw = obs > tgt + db;
        bool dn_raw = obs + db < tgt;
        bool fb_inc = (mode == 6) ? dn_raw : up_raw;
        bool fb_dec = (mode == 6) ? up_raw : dn_raw;

        int64_t up = (int64_t)b + (int64_t)step;
        uint64_t b_inc_val = (up > (int64_t)maxbud) ? maxbud : (uint64_t)up;
        int64_t dn = (int64_t)b - (int64_t)step;
        uint64_t b_dec_val = (dn < (int64_t)minbud) ? minbud : (uint64_t)dn;
        bool apply_fb = win_complete && mode != 0 && (fb_inc || fb_dec);
        uint64_t b_next = apply_fb ? (fb_inc ? b_inc_val : b_dec_val) : b;

        // ---- mode6 countdown ----------------------------------------------
        uint64_t cd_dec = (win_run && m6_cd != 0) ? m6_cd - 1 : m6_cd;
        uint64_t cd_next = cd_dec;
        bool gap_load = false;
        bool m6_started_next = m6_started;
        if (win_run && mode == 6 && in.a_done) {
            if (m6_started) { cd_next = b; gap_load = true; }
            else            { m6_started_next = true; }
        }

        // ---- statistics increments -----------------------------------------
        uint64_t inc[26] = {0};
        uint64_t hinc[32] = {0};
        inc[ST_ARMEDCYC] = win_run ? 1 : 0;
        inc[ST_CHARGE]   = charge_now;
        inc[ST_CHARGECYC]= (charge_now != 0) ? 1 : 0;
        inc[ST_REPLEN]   = refill_now;
        inc[ST_VDONE]    = (win_run && in.v_done)  ? 1 : 0;
        inc[ST_VBYTES]   = (win_run && in.v_done)  ? 4096 : 0;
        inc[ST_AISSUE]   = (win_run && in.a_issue) ? 1 : 0;
        inc[ST_ADONE]    = (win_run && in.a_done)  ? 1 : 0;
        inc[ST_HIT]      = (win_run && ev_hit_a)   ? 1 : 0;
        inc[ST_MISS]     = (win_run && ev_miss_a)  ? 1 : 0;
        inc[ST_PTW]      = (win_run && ev_ptw_a)   ? 1 : 0;
        inc[ST_AOWN]     = (win_run && ev_aown)    ? 1 : 0;
        inc[ST_SEL]      = (win_run && ev_sel_a)   ? 1 : 0;
        inc[ST_LUTA]     = (win_run && ev_lut_a)   ? 1 : 0;
        inc[ST_LUTAMISS] = (win_run && ev_miss_a)  ? 1 : 0;
        inc[ST_ACTUALCYC]= (win_run && ev_act_a)   ? 1 : 0;
        inc[ST_WIN]      = win_complete ? 1 : 0;
        inc[ST_WINALL]   = win_last ? 1 : 0;
        inc[ST_WINPART]  = (win_last && !act_all) ? 1 : 0;
        inc[ST_GAPL]     = gap_load ? 1 : 0;
        inc[ST_STALE]    = stale ? 1 : 0;
        bool shortfall = win_complete && (obs < tgt);
        inc[ST_SHORTW]   = shortfall ? 1 : 0;
        inc[ST_SHORTB]   = shortfall ? (uint64_t)(tgt - obs) : 0;
        inc[ST_FBUP]     = (win_complete && mode != 0 && fb_inc) ? 1 : 0;
        inc[ST_FBDN]     = (win_complete && mode != 0 && fb_dec) ? 1 : 0;

        bool cnt_ovf = false;
        uint64_t m = mask();
        uint64_t nstat[26];
        for (int i = 0; i < 26; i++) {
            unsigned __int128 s = (unsigned __int128)stat[i] + inc[i];
            if (s > m) cnt_ovf = true;
            nstat[i] = (uint64_t)s & m;
        }
        uint64_t nhist[32];
        for (int i = 0; i < 32; i++) nhist[i] = hist[i];
        if (win_complete) {
            int hb = (vdone_now >= 31) ? 31 : (int)vdone_now;
            unsigned __int128 s = (unsigned __int128)hist[hb] + 1;
            if (s > m) cnt_ovf = true;
            nhist[hb] = (uint64_t)s & m;
        }

        // ---- write decode ---------------------------------------------------
        bool f_arm = false, f_abort = false, f_freeze = false, f_clear = false, f_cfg = false;
        uint64_t ne = do_wr ? decode_write(widx, wdata, in.quiet,
                                           f_arm, f_abort, f_freeze, f_clear, f_cfg)
                            : 0;

        // ---- outputs (combinational, pre-edge state) -------------------------
        bool obs_err_now  = armed && !frozen && in.response_error;
        bool fatal_reg    = (err & FATAL_MASK) != 0;
        bool fatal_now    = fatal_reg || obs_err_now || cnt_ovf || bal_ovf;
        bool enable_core  = enabled && armed && !frozen;
        bool allow_core   = (mode == 0) ? true
                          : (mode == 6) ? (cd_next == 0)
                                        : (bal_next >= 0);
        bool blocked      = enable_core && (aborted || !allow_core || fatal_reg);

        RefObs o;
        o.en    = enable_core;
        o.allow = enable_core && !aborted && !fatal_now && allow_core;
        o.abort = enable_core && (aborted || fatal_now);

        // ---- commit ---------------------------------------------------------
        inc[ST_BLOCKED] = blocked ? 1 : 0;
        {   unsigned __int128 s = (unsigned __int128)stat[ST_BLOCKED] + inc[ST_BLOCKED];
            if (s > m) cnt_ovf = true;
            nstat[ST_BLOCKED] = (uint64_t)s & m;
        }

        if (f_clear) {
            std::memset(stat, 0, sizeof(stat));
            std::memset(hist, 0, sizeof(hist));
            err = 0;
            armed = frozen = aborted = false;
            balance = (int64_t)init_clamped();
            b = bmin = bmax = init_clamped();
            prev_main = win_main = 0;
            win_cnt = win_vdone = 0;
            win_allact = false;
            m6_started = false; m6_cd = 0;
            nodone_run = nodone_max = 0;
        } else {
            for (int i = 0; i < 26; i++) stat[i] = nstat[i];
            for (int i = 0; i < 32; i++) hist[i] = nhist[i];
            err |= ne;
            if (obs_err_now) err |= (1ull << E_OBS);
            if (cnt_ovf)     err |= (1ull << E_CNTOVF);
            if (bal_ovf)     err |= (1ull << E_BALOVF);
        }

        if (f_cfg && !armed) set_cfg(widx, (uint32_t)wdata);
        if (f_arm && !f_clear) arm_now();
        if (f_abort) aborted = true;
        if (f_freeze) frozen = true;

        // datapath (never concurrent with a command, which needs !armed)
        if (win_run && !f_arm && !f_clear) {
            balance = bal_next;
            if (win_last) { win_cnt = 0; win_allact = true; win_vdone = 0; }
            else          { win_cnt++; win_allact = act_all; win_vdone = vdone_now; }
            if (win_last) { prev_main = win_main + (mode == 7 && ev_aown ? 1 : 0);
                            win_main = 0; }
            else          { win_main += (mode == 7 && ev_aown ? 1 : 0); }
            if (apply_fb) {
                b = b_next;
                if (b < bmin) bmin = b;
                if (b > bmax) bmax = b;
            }
            m6_cd = cd_next;
            m6_started = m6_started_next;
        }

        {
            bool act_now = win_run && in.v_active && !in.v_done;
            nodone_run = act_now ? nodone_run + 1 : 0;
            if (nodone_run > nodone_max) nodone_max = nodone_run;
        }
        return o;
    }
};

static void cmp_obs(const Obs &a, const RefObs &b, const char *where) {
    if (a.en != b.en || a.allow != b.allow || a.abort != b.abort) {
        char msg[256];
        std::snprintf(msg, sizeof(msg),
                      "%s: rtl en=%d allow=%d abort=%d ; ref en=%d allow=%d abort=%d",
                      where, a.en, a.allow, a.abort, b.en, b.allow, b.abort);
        throw std::runtime_error(msg);
    }
    checks++;
}

// ---------------------------------------------------------------------------
// directed helpers
// ---------------------------------------------------------------------------
static void cfg(Ref &r, Bench &b, uint32_t idx, uint32_t v) {
    b.wr(idx, v);
    r.set_cfg(idx, v);
}

static void cmd(Bench &b, Ref &r, uint32_t c) {
    b.wr(R_COMMAND, c);
    bool a, ab, f, cl, cg;
    uint64_t ne = r.decode_write(R_COMMAND, c, b.cur.quiet, a, ab, f, cl, cg);
    if (cl) r.clear_now();
    else {
        r.err |= ne;
        if (a) r.arm_now();
        if (ab) r.aborted = true;
        if (f) r.frozen = true;
    }
}

static void t_bad_writes_and_sticky(Bench &b, Ref &r) {
    b.reset(); r.do_reset();

    b.wr(R_MODE, 0x100000000ull);                    // high bits nonzero
    check((b.rd(R_ERROR) >> E_BADADDR) & 1, "high bits sticky");
    b.wr(R_COMMAND, 99);                             // unknown command

    b.apply(In()); b.cur.quiet = true;
    b.wr(R_COMMAND, 4);
    check(b.rd(R_ERROR) == 0, "legal CLEAR clears sticky errors");
    b.apply(In());
}

static void t_mode1_charge_and_debt(Bench &b, Ref &r, bool narrow) {
    b.reset(); r.do_reset(); r.cw = narrow ? 8 : 64;
    In idle; idle.quiet = true;
    b.apply(idle);
    cfg(r, b, R_MODE, 1);
    cfg(r, b, R_WINDOW, 64);
    cfg(r, b, R_INITBUD, 4);          // tiny budget -> debt after 2 issues
    cfg(r, b, R_MINBUD, 1);
    cfg(r, b, R_MAXBUD, 32768);
    cfg(r, b, R_STEP, 512);
    cfg(r, b, R_CAP, 65536);
    cfg(r, b, R_TARGET, 0);
    cfg(r, b, R_DEADBAND, 0);
    cmd(b, r, 1);                     // ARM

    In in; in.quiet = true;
    // first cycle: balance 4 >= 0 -> allow
    Obs o = b.drive(in);
    check(o.en && o.allow, "first admit allowed with positive budget");
    b.edge();

    // one A issue charges 4096 -> balance deeply negative, allow drops
    in.a_issue = true;
    o = b.drive(in);
    check(o.en && !o.allow, "debt blocks further admission");
    b.edge();
    in.a_issue = false;

    for (int i = 0; i < 70; i++) { b.drive(in); b.edge(); }
    o = b.drive(in);
    check(!o.allow, "debt not forgiven by window refill");
    int64_t bal = (int64_t)b.rd(S_BALANCE);
    check(bal < 0, "negative balance preserved");
    check(b.rd(S_CHARGE) == 4096, "single 4096 charge");
    b.apply(In()); b.cur.quiet = true; b.wr(R_COMMAND, 4);
}

static void t_window_feedback(Bench &b, Ref &r) {
    b.reset(); r.do_reset();
    In idle; idle.quiet = true; b.apply(idle);
    cfg(r, b, R_MODE, 1);
    cfg(r, b, R_WINDOW, 8);
    cfg(r, b, R_INITBUD, 8000);
    cfg(r, b, R_MINBUD, 1);
    cfg(r, b, R_MAXBUD, 32768);
    cfg(r, b, R_STEP, 512);
    cfg(r, b, R_CAP, 65536);
    cfg(r, b, R_TARGET, 8192);
    cfg(r, b, R_DEADBAND, 0);
    cmd(b, r, 1);

    // one fully V-active window with zero V done -> b up, bucket 0
    In in; in.quiet = true; in.v_active = true;
    for (int i = 0; i < 8; i++) { b.drive(in); b.edge(); }
    check(b.rd(S_WIN) == 1, "one complete active window");
    check(b.rd(S_HIST + 0) == 1, "histogram bucket 0");
    check(b.rd(S_FBUP) == 1 && b.rd(S_FBDN) == 0, "observed below target -> b up");
    check(b.rd(S_BFINAL) == 8000 + 512, "b stepped up by 512");
    check(b.rd(S_SHORTW) == 1 && b.rd(S_SHORTB) == 8192, "shortfall recorded");

    // one fully V-active window with 8 V done (32768 B > target) -> b down
    In busy = in; busy.v_done = true;
    for (int i = 0; i < 8; i++) { b.drive(busy); b.edge(); }
    check(b.rd(S_WIN) == 2, "two complete active windows");
    check(b.rd(S_HIST + 8) == 1, "histogram bucket 8");
    check(b.rd(S_FBUP) == 1 && b.rd(S_FBDN) == 1, "b stepped down once");
    check(b.rd(S_REPLEN) == 8000 + (8000 + 512), "two refills from the pre-window b");

    // partial window (one idle cycle) must not update b
    uint64_t before = b.rd(S_BFINAL);
    In gap = in; gap.v_active = false;
    b.drive(gap); b.edge();
    for (int i = 1; i < 8; i++) { b.drive(in); b.edge(); }
    check(b.rd(S_WINPART) == 1, "partial window counted separately");
    check(b.rd(S_BFINAL) == before, "partial window does not update b");
    b.apply(In()); b.cur.quiet = true; b.wr(R_COMMAND, 4);
}

static void t_mode6_gap(Bench &b, Ref &r) {
    b.reset(); r.do_reset();
    In idle; idle.quiet = true; b.apply(idle);
    cfg(r, b, R_MODE, 6);
    cfg(r, b, R_WINDOW, 64);
    cfg(r, b, R_INITBUD, 3);
    cfg(r, b, R_MINBUD, 0);
    cfg(r, b, R_MAXBUD, 32768);
    cmd(b, r, 1);

    In in; in.quiet = true;
    Obs o = b.drive(in);
    check(o.en && o.allow, "mode6 first admit allowed (countdown 0)");
    b.edge();

    in.a_done = true;                       // first A done: no gap
    o = b.drive(in);
    check(o.allow, "first A done inserts no gap");
    b.edge();

    o = b.drive(in);                        // second A done: loads gap 3
    check(o.allow, "countdown still zero before load");
    b.edge();

    in.a_done = true;
    o = b.drive(in);
    check(!o.allow, "decision uses the updated countdown");
    b.edge();
    in.a_done = false;

    o = b.drive(in); check(!o.allow, "countdown 3 -> blocked"); b.edge(); b.edge();
    o = b.drive(in); check(!o.allow, "countdown 1 -> blocked"); b.edge();
    o = b.drive(in); check(o.allow, "countdown exhausted -> allow");
    check(b.rd(S_GAPL) == 1, "exactly one gap load");
    b.apply(In()); b.cur.quiet = true; b.wr(R_COMMAND, 4);
}

    b.apply(In()); b.cur.quiet = true; b.wr(R_COMMAND, 4);
}

static void t_error_abort_and_recovery(Bench &b, Ref &r) {
    b.reset(); r.do_reset();
    In in; in.quiet = true; b.apply(in);
    cmd(b, r, 1);
    check(b.rd(R_STATUS) & 1, "armed");

    in.response_error = true;
    Obs o = b.drive(in);
    check(!o.allow && o.abort, "observation error blocks and aborts");
    b.edge();
    in.response_error = false;
    o = b.drive(in);
    check(!o.allow && o.abort, "observation error is sticky");
    check((b.rd(R_ERROR) >> E_OBS) & 1, "E_OBS sticky");

    b.apply(In()); b.cur.quiet = true;
    b.wr(R_COMMAND, 3);                     // FREEZE requires quiet
    check(b.rd(R_STATUS) & 2, "frozen");
    b.wr(R_COMMAND, 4);                     // CLEAR needs !armed -> illegal
    check((b.rd(R_ERROR) >> E_CLEARILL) & 1, "clear while armed is illegal");
    check(b.rd(R_ERROR) != 0, "illegal clear cannot clear errors");
}

static void t_abort_keeps_charging(Bench &b, Ref &r) {
    b.reset(); r.do_reset();
    In in; in.quiet = true; b.apply(in);
    cfg(r, b, R_MODE, 1);
    cfg(r, b, R_WINDOW, 32);
    cmd(b, r, 1);
    cmd(b, r, 2);                            // ABORT
    Obs o = b.drive(in);
    check(o.en && !o.allow && o.abort, "abort stops new admission");
    b.edge();
    in.a_issue = true;
    b.drive(in);
    b.edge();
    in.a_issue = false;
    check(b.rd(S_CHARGE) == 4096, "in-flight charging continues after ABORT");
    b.apply(In()); b.cur.quiet = true; b.wr(R_COMMAND, 3);   // freeze drains
    check(b.rd(R_STATUS) & 2, "freeze after abort");
    b.wr(R_COMMAND, 4);
}

static void t_narrow_overflow(Bench &b) {
    b.reset();
    In in; in.quiet = true; b.apply(in);
    b.wr(R_MODE, 0);
    b.wr(R_WINDOW, 1);
    b.wr(R_COMMAND, 1);                      // ARM

    for (int i = 0; i < 300; i++) { b.drive(in); b.edge(); }  // > 256 armed cycles
    check((b.rd(R_ERROR) >> E_CNTOVF) & 1, "CounterWidth overflow sticky");
    check(b.rd(S_ARMEDCYC) == (300 & 0xFF), "statistic wraps at CounterWidth");
    Obs o = b.drive(in);
    check(!o.allow, "fatal overflow blocks allow even in mode 0");
    check(o.abort, "fatal overflow raises abort");

    b.apply(In()); b.cur.quiet = true;
    b.wr(R_COMMAND, 4);                      // legal CLEAR recovers
    check(b.rd(R_ERROR) == 0, "CLEAR clears counter overflow");
    check(b.rd(S_ARMEDCYC) == 0, "CLEAR zeroes statistics");
}

static void t_enabled0(Bench &b) {
    b.reset();
    check(b.rd(R_MAGIC) == 0x49444D414D435431ull, "Enabled=0 magic still readable");
    check(b.rd(R_VERSION) == 1, "Enabled=0 version still readable");
    In in; in.quiet = true; in.supported = true;
    b.apply(in);
    b.wr(R_COMMAND, 1);
    check((b.rd(R_ERROR) >> E_UNSUP) & 1, "ARM refused as unsupported");
    check(b.rd(R_STATUS) == 0, "never armed");
    Obs o = b.drive(in);
    check(!o.en && !o.allow && !o.abort, "feature absent: outputs inactive");
}

static void run_reference(const char *tag, uint32_t seed, int cycles,
                          bool narrow, int mode) {
    Bench b;
    Ref   r;
    r.cw = narrow ? 8 : 64;

    In idle; idle.quiet = true;
    b.apply(idle);
    cfg(r, b, R_MODE, (uint32_t)mode);
    cfg(r, b, R_WINDOW, narrow ? 7 : 23);
    cfg(r, b, R_TARGET, 12000);
    cfg(r, b, R_DEADBAND, 4096);
    cfg(r, b, R_INITBUD, 5000);
    cfg(r, b, R_MINBUD, narrow ? 1 : (mode == 6 ? 0 : 1));
    cfg(r, b, R_MAXBUD, 32768);
    cfg(r, b, R_STEP, narrow ? 3 : 1024);
    cfg(r, b, R_CAP, 65536);
    cfg(r, b, R_WHIT, 1);
    cfg(r, b, R_WMISS, 100);
    cfg(r, b, R_WPTW, 100);
    cmd(b, r, 1);

    std::mt19937 rng(seed);
    std::uniform_int_distribution<int> bite(0, 99);
    In in;
    bool quiet_phase = false;

    for (int c = 0; c < cycles; c++) {
        in = In();
        in.quiet     = quiet_phase;
        in.supported = true;
        in.v_active  = bite(rng) < 80;
        in.a_issue   = bite(rng) < 15;
        in.a_done    = bite(rng) < 10;
        in.v_done    = bite(rng) < 40;
        in.ctl_did   = (bite(rng) < 70) ? DID_A : (uint32_t)(0x11 + (c % 3));
        in.ctl_state = (uint32_t)((bite(rng) < 50) ? 1 : (bite(rng) < 60 ? 2 : 3));
        in.select_fire = bite(rng) < 20;
        in.select_did  = (bite(rng) < 70) ? DID_A : 0x22;
        in.lookup      = bite(rng) < 25;
        in.lookup_hit  = bite(rng) < 50;
        in.lookup_did  = (bite(rng) < 70) ? DID_A : 0x22;
        in.ptw_r_fire  = bite(rng) < 12;
        in.response_error = bite(rng) < 2;

        bool do_wr = false;
        uint32_t widx = 0;
        uint64_t wdata = 0;
        int roll = bite(rng);
        if (roll < 3) {                 // random config write (usually locked)
            do_wr = true;
            widx  = R_STEP;
            wdata = (uint64_t)(1 + (rng() % 1000));
        } else if (roll < 4) {          // random bad address
            do_wr = true;
            widx  = 0x77;
            wdata = 0;
        } else if (roll == 5) {         // legal CLEAR needs quiet && !armed
            r.err |= 0;                 // nothing to mirror before the write
            b.apply(in);
            // CLEAR is only legal when quiet && !armed; in this run the DUT is
            // armed, so this is the expected "illegal clear" sticky path.
            b.wr(R_COMMAND, 4);
            bool fa, fb_, ff, fc, fg;
            uint64_t ne = r.decode_write(R_COMMAND, 4, in.quiet, fa, fb_, ff, fc, fg);
            r.err |= ne;
            b.apply(in);
            continue;
        }
        if (bite(rng) < 2) {            // ABORT
            b.apply(in);
            b.wr(R_COMMAND, 2);
            bool fa, fb_, ff, fc, fg;
            uint64_t ne = r.decode_write(R_COMMAND, 2, in.quiet, fa, fb_, ff, fc, fg);
            if (fc) r.clear_now(); else { r.err |= ne; if (fa) r.arm_now();
                                          if (fb_) r.aborted = true; if (ff) r.frozen = true; }
            b.apply(in);
            continue;
        }

        b.apply(in);
        b.d.cfg_valid_i = do_wr ? 1 : 0;
        b.d.cfg_write_i = do_wr ? 1 : 0;
        b.d.index_i     = (uint16_t)widx;
        b.d.wdata_i     = wdata;
        b.d.clk_i = 0; b.d.eval();

        Obs rtl{ (bool)b.d.policy_enable_o, (bool)b.d.policy_allow_o,
                 (bool)b.d.policy_abort_o };
        RefObs ref = r.cycle(in, do_wr, widx, wdata);

        if (rtl.en != ref.en || rtl.allow != ref.allow || rtl.abort != ref.abort) {
            char msg[256];
            std::snprintf(msg, sizeof(msg),
                "%s cycle %d en/allow/abort rtl=%d/%d/%d ref=%d/%d/%d",
                tag, c, rtl.en, rtl.allow, rtl.abort, ref.en, ref.allow, ref.abort);
            throw std::runtime_error(msg);
        }
        checks++;

        b.d.clk_i = 1; b.d.eval(); b.d.clk_i = 0; b.d.eval();
        b.d.cfg_valid_i = 0; b.d.cfg_write_i = 0; b.d.wdata_i = 0;

        b.apply(in);
    }

    check(b.rd(R_ERROR) == r.err, "reference error register");
    check(b.rd(S_CHARGE) == (r.stat[Ref::ST_CHARGE] & r.mask()), "reference total charge");
    check(b.rd(S_REPLEN) == (r.stat[Ref::ST_REPLEN] & r.mask()), "reference total replenish");
    check(b.rd(S_WIN) == (r.stat[Ref::ST_WIN] & r.mask()), "reference complete windows");
    check(b.rd(S_VDONE) == (r.stat[Ref::ST_VDONE] & r.mask()), "reference V done events");
    check(b.rd(S_FBUP) == (r.stat[Ref::ST_FBUP] & r.mask()), "reference b-up count");
    check(b.rd(S_FBDN) == (r.stat[Ref::ST_FBDN] & r.mask()), "reference b-down count");
    check(b.rd(S_BLOCKED) == (r.stat[Ref::ST_BLOCKED] & r.mask()), "reference blocked cycles");
    check((int64_t)b.rd(S_BALANCE) == r.balance, "reference final balance");
    check(b.rd(S_BFINAL) == r.b, "reference final b");
    check(b.rd(S_BMIN) == r.bmin, "reference b min");
    check(b.rd(S_BMAX) == r.bmax, "reference b max");
    check(b.rd(S_NODONE) == r.nodone_max, "reference longest no-done run");
    for (int i = 0; i < 32; i++)
        check(b.rd(S_HIST + i) == (r.hist[i] & r.mask()), "reference histogram bucket");
}

// ---------------------------------------------------------------------------
int main(int argc, char **argv) {
    Verilated::commandArgs(argc, argv);
    std::string mode = (argc > 1) ? std::string(argv[1]) : std::string("full");

    try {
        if (mode == "off") {
            Bench b;
            t_enabled0(b);
        } else if (mode == "small") {
            Bench b;
            t_narrow_overflow(b);
            run_reference("small-ref-m0", 0xC0FFEE01u, 260, true, 0);
            run_reference("small-ref-m3", 0xC0FFEE02u, 260, true, 3);
        } else {
            Bench b;
            Ref   r;
            t_reset_defaults(b);
            t_bad_writes_and_sticky(b, r);
            t_arm_predicates(b);
            t_config_lock(b);
            t_first_allow_and_mode0(b);
            t_mode1_charge_and_debt(b, r, false);
            t_window_feedback(b, r);
            t_mode6_gap(b, r);
            t_mode7_stale(b, r);
            t_error_abort_and_recovery(b, r);
            t_abort_keeps_charging(b, r);

            for (int m = 0; m <= 7; m++) {
                char nm[32];
                std::snprintf(nm, sizeof(nm), "ref-mode%d", m);
                run_reference(nm, 0x12345678u + (uint32_t)m, 400, false, m);
            }
        }
        std::cout << "PASS " << checks
                  << " dma_research_policy checks (simulation only)" << std::endl;
        return 0;
    } catch (const std::exception &e) {
        std::cerr << "FAIL after " << checks << " checks: " << e.what() << std::endl;
        return 1;
    }
}
